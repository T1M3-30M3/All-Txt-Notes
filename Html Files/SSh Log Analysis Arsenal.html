# Real-time SSH monitoring
tail -f /var/log/auth.log | grep sshd

📊 2. Brute Force Detection & Analysis

Failed login attempts (top attackers)

grep "Failed password" /var/log/auth.log | awk '{print $11}' | sort | uniq -c | sort -nr | head -20

IP-based attack analysis

grep "Failed password.*for .* from" /var/log/auth.log | awk '{print $11}' | sort | uniq -c | sort -nr > ssh_attackers.txt

Successful logins after brute force

grep -E "Accepted password|Accepted publickey" /var/log/auth.log | grep -A5 -B5 "Failed password"

Time-based attack windows

grep "Failed password" /var/log/auth.log | cut -d' ' -f1-3 | sort | uniq -c | sort -nr | head

GeoIP analysis of attackers

awk '{print $11}' ssh_attackers.txt | while read ip; do
    geoiplookup $ip 2>/dev/null | head -1
done | sort | uniq -c | sort -nr

🎯 3. Advanced SSH Pattern Analysis

User enumeration attempts

grep "Invalid user" /var/log/auth.log | awk '{print $9}' | sort | uniq -c | sort -nr

Key-based auth analysis

grep "Accepted publickey" /var/log/auth.log | awk '{print $9" "$11" "$13}' | sort | uniq -c

PAM module failures

grep "pam_unix" /var/log/auth.log | grep "authentication failure" | awk '{print $11}' | sort | uniq -c

Complete SSH session timeline

awk '/sshd/ && /from/ {print $1" "$2" "$9" "$11}' /var/log/auth.log | sort > ssh_timeline.txt

🤖 4. Complete SSH Analyzer Script

#!/bin/bash
LOGFILE=${1:-/var/log/auth.log}
echo "=== SSH ATTACK ANALYSIS ==="
echo "Top 20 Attackers:"
grep "Failed password.*from" $LOGFILE | awk '{print $11}' | sort | uniq -c | sort -nr | head -20

echo -e "\nSuccessful Logins (Last 24h):"
grep "Accepted" $LOGFILE | tail -n 50

echo -e "\nUser Enumeration Attempts:"
grep "Invalid user" $LOGFILE | awk '{print $9}' | sort | uniq -c | sort -nr

echo -e "\nAttack Timeline (hourly):"
grep "Failed password" $LOGFILE | cut -d' ' -f1-2 | cut -d: -f1 | sort | uniq -c | sort -nr
