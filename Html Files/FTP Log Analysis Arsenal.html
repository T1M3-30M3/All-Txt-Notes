🔐 2. FTP Brute Force & Access Analysis

Failed login attempts

grep "FAIL LOGIN" /var/log/vsftpd.log | awk '{print $7" "$9}' | sort | uniq -c | sort -nr

Successful logins

grep "OK LOGIN" /var/log/vsftpd.log | awk '{print $7" "$9" "$10}' | sort | uniq -c

Anonymous access

grep "anon" /var/log/vsftpd.log | grep "OK LOGIN" | wc -l
grep "anonymous" /var/log/xferlog | awk '{print $2" "$9}' | sort | uniq

Download/upload analysis (xferlog format)

# Total Upload in MB
awk '$9=="o" {sum+=10*$7/1024} END {print "Total Upload (MB): "sum}' /var/log/xferlog

# Total Download in MB
awk '$9=="i" {sum+=10*$7/1024} END {print "Total Download (MB): "sum}' /var/log/xferlog

🎯 3. Advanced FTP Behavioral Analysis

Directory traversal attempts

grep -i "\/..\/" /var/log/vsftpd.log | awk '{print $7" attempted: "$NF}'

Long sessions (potential exfil)

awk '$5-$4 > 3600 {print $0}' /var/log/xferlog | wc -l

Top downloaded files

grep "i" /var/log/xferlog | awk '{print $9" "$2}' | sort | uniq -c | sort -nr | head -20

Chroot escapes

grep -i "chroot" /var/log/vsftpd.log | grep -v "OK"

🤖 4. Complete FTP Analyzer Script

#!/bin/bash
LOGFILE=${1:-/var/log/vsftpd.log}
echo "=== FTP ACCESS ANALYSIS ==="
echo "Failed Logins:"
grep "FAIL LOGIN" $LOGFILE | awk '{print $7" "$9}' | sort | uniq -c | sort -nr | head -15

echo -e "\nSuccessful Logins:"
grep "OK LOGIN" $LOGFILE | awk '{print $7" "$9" "$10}' | sort | uniq -c | head -15

echo -e "\nAnon Access:"
grep -i anon $LOGFILE | grep "OK LOGIN" | wc -l

echo -e "\nTop Downloads:"
grep "i 1" $LOGFILE | awk '{print $NF" ("$7" bytes)"}' | sort | uniq -c | sort -nr | head -10
